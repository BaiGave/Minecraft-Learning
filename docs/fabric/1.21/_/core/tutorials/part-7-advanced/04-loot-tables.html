1. 战利品表概述

1.1 什么是战利品表？

战利品表（Loot Table）定义了实体死亡、方块破坏或箱子打开时掉落的物品。Fabric 提供了 fabric-loot-api-v3 模块，允许我们在运行时修改战利品表。

1.2 依赖

dependencies {
    // 战利品表 API（包含在 fabric-api 中）
    modImplementation "net.fabricmc:fabric-api:${fabric_version}"
}

2. LootTableEvents 修改器

2.1 修改事件

import net.fabricmc.fabric.api.loot.v3.LootTableEvents;

// 注册战利品表修改事件
LootTableEvents.MODIFY.register((key, tableBuilder, source, registries) => {
    // key: LootTableKey - 战利品表的标识
    // tableBuilder: LootTable.Builder - 战利品表构建器
    // source: LootTableSource - 来源（内置/数据包/模组）
    // registries: RegistryWrapper.BuiltinRegistry - 注册表访问
    
    // 修改逻辑
});

2.2 过滤来源

LootTableEvents.MODIFY.register((key, tableBuilder, source, registries) => {
    // 只修改内置战利品表（游戏原版）
    if (source.isBuiltin()) {
        // 修改内容
    }
    
    // 检查特定战利品表
    if (key.equals(LootTables.ENTITIES.CREEPER)) {
        // 修改苦力怕掉落
    }
});

3. 战利品表池与条件

3.1 LootPool 战利品池

import net.minecraft.loot.LootPool;
import net.minecraft.loot.entry.ItemEntry;
import net.minecraft.loot.entry.LootPoolEntry;
import net.minecraft.loot.condition.LootCondition;

// 创建战利品池
LootPool.Builder pool = LootPool.builder()
    .rolls(ConstantLootNumberProvider.create(1))  // 掉落次数
    .bonusRolls(ConstantLootNumberProvider.create(0))  // 额外掉落
    .with(ItemEntry.builder(Items.DIAMOND))  // 添加物品
    .conditionally(LootCondition.Builder...);  // 添加条件

3.2 常用条件

| 条件 | 说明 | |-----|------| | SurvivesExplosion | 仅在爆炸中存活时触发 | | EntityProperties | 检查实体属性 | | KilledByPlayer | 被玩家击杀 | | RandomChance | 随机概率 | | TableBonus | 抢夺附魔加成 | | Inverted | 条件反转 |

3.3 条件示例

import net.minecraft.loot.condition.RandomChanceLootCondition;
import net.minecraft.loot.condition.TableBonusLootCondition;

// 10% 概率掉落
.randomChance(0.1f)

// 抢夺附魔加成（基于附魔等级）
.tableBonus(Enchantments.FORTUNE, 0.1f)

// 被玩家击杀
.killedByPlayer()

// 组合条件
.and(
    RandomChanceLootCondition.randomChance(0.5f),
    KilledByPlayerLootCondition.killedByPlayer()
)

4. 完整示例：修改方块掉落

4.1 修改原石掉落

package net.example.mymod.loot;

import net.fabricmc.fabric.api.loot.v3.LootTableEvents;
import net.minecraft.enchantment.Enchantments;
import net.minecraft.loot.LootPool;
import net.minecraft.loot.LootTables;
import net.minecraft.loot.condition.LootCondition;
import net.minecraft.loot.condition.SurvivesExplosionLootCondition;
import net.minecraft.loot.condition.TableBonusLootCondition;
import net.minecraft.loot.entry.ItemEntry;

public class BlockLootModifier {
    
    public static void register() {
        // 修改圆石掉落：添加铁锭（5% 基础概率，有抢夺时增加）
        LootTableEvents.MODIFY.register((key, tableBuilder, source, registries) -> {
            // 只修改原版圆石掉落
            if (source.isBuiltin() && key.equals(LootTables.BLOCKS.COBBLESTONE)) {
                // 创建新池
                LootPool.Builder pool = LootPool.builder()
                    // 基础掉落：圆石
                    .with(ItemEntry.builder(net.minecraft.block.Blocks.COBBLESTONE)
                        .weight(1)
                    )
                    // 铁锭掉落（5% 基础概率）
                    .with(ItemEntry.builder(Items.IRON_INGOT)
                        .weight(1)
                        .conditionally(
                            // 爆炸存活 + 抢夺附魔加成
                            LootCondition.and(
                                SurvivesExplosionLootCondition.builder(),
                                TableBonusLootCondition.builder(
                                    registries.getOrThrow(net.minecraft.registry.Registry.ENCHANTMENT_KEY)
                                        .getOrThrow(Enchantments.FORTUNE),
                                    0.05f  // 基础 5%
                                )
                            )
                        )
                    )
                    // 金锭掉落（1% 基础概率）
                    .with(ItemEntry.builder(Items.GOLD_INGOT)
                        .weight(1)
                        .conditionally(
                            LootCondition.and(
                                SurvivesExplosionLootCondition.builder(),
                                TableBonusLootCondition.builder(
                                    registries.getOrThrow(net.minecraft.registry.Registry.ENCHANTMENT_KEY)
                                        .getOrThrow(Enchantments.FORTUNE),
                                    0.01f
                                )
                            )
                        )
                    );
                
                // 清除原有池并添加新池
                tableBuilder.pool(pool);
            }
        });
    }
}

                    )

    }
});

4.3 修改钻石矿石掉落

LootTableEvents.MODIFY.register((key, tableBuilder, source, registries) -> {

                .weight(1)

                    ConstantLootNumberProvider.create(1)
                ))
            );
    }
});

5. 完整示例：修改实体掉落

5.1 修改僵尸掉落

package net.example.mymod.loot;

import net.fabricmc.fabric.api.loot.v3.LootTableEvents;
import net.minecraft.enchantment.Enchantments;
import net.minecraft.loot.LootPool;
import net.minecraft.loot.LootTables;
import net.minecraft.loot.condition.LootCondition;
import net.minecraft.loot.condition.RandomChanceLootCondition;
import net.minecraft.loot.condition.TableBonusLootCondition;
import net.minecraft.loot.entry.ItemEntry;
import net.minecraft.registry.RegistryKeys;

public class EntityLootModifier {
    
    public static void register() {
        // 修改僵尸掉落
        LootTableEvents.MODIFY.register((key, tableBuilder, source, registries) -> {
            if (source.isBuiltin() && key.equals(LootTables.ENTITIES.ZOMBIE)) {
                // 保留原有掉落并添加自定义
                LootPool.Builder pool = LootPool.builder()
                    // 腐肉（原有）
                    .with(ItemEntry.builder(Items.ROTTEN_FLESH)
                        .weight(1)
                        .apply(SetCountLootFunction.builder(
                            UniformLootNumberProvider.create(0f, 2f)
                        ))
                    )
                    // 铁锭（5% 概率）
                    .with(ItemEntry.builder(Items.IRON_INGOT)
                        .weight(1)
                        .conditionally(
                            RandomChanceLootCondition.randomChance(0.05f)
                        )
                    )
                    // 马铃薯（10% 概率，仅在沼泽生物群系）
                    .with(ItemEntry.builder(Items.POTATO)
                        .weight(1)
                        .conditionally(
                            RandomChanceLootCondition.randomChance(0.1f)
                        )
                    )
                    // 抢夺附魔加成
                    .with(ItemEntry.builder(Items.GOLD_INGOT)
                        .weight(1)
                        .conditionally(
                            TableBonusLootCondition.builder(
                                registries.getOrThrow(RegistryKeys.ENCHANTMENT)
                                    .getOrThrow(Enchantments.LOOTING),
                                0.01f  // 基础1%
                            )
                        )
                    );
                
                tableBuilder.pool(pool);
            }
        });
    }
}

6. 完整示例：添加自定义掉落

6.1 创建自定义战利品表

package net.example.mymod.loot;

import net.fabricmc.fabric.api.loot.v3.LootTableSource;
import net.minecraft.util.Identifier;
import net.minecraft.world.loot.LootTable;
import net.minecraft.world.loot.LootTableManager;
import net.minecraft.registry.RegistryKey;
import net.minecraft.registry.RegistryKeys;

// 在 modid 命名空间下创建
public class ModLootTables {
    public static final RegistryKey<LootTable> CUSTOM_CHEST = 
        RegistryKey.of(RegistryKeys.LOOT_TABLE, Identifier.of("mymod", "chests/custom_chest"));
    
    // 注册自定义战利品表
    public static void register() {
        // 创建战利品表
        LootTable table = LootTable.builder()
            .pool(LootPool.builder()
                .rolls(ConstantLootNumberProvider.create(3))
                .with(ItemEntry.builder(Items.DIAMOND)
                    .weight(1)
                    .apply(SetCountLootFunction.builder(
                        UniformLootNumberProvider.create(1f, 3f)
                    ))
                )
                .with(ItemEntry.builder(Items.GOLD_INGOT)
                    .weight(2)
                    .apply(SetCountLootFunction.builder(
                        UniformLootNumberProvider.create(2f, 5f)
                    ))
                )
            )
            .pool(LootPool.builder()
                .rolls(ConstantLootNumberProvider.create(1))
                .with(ItemEntry.builder(Items.ENCHANTED_BOOK)
                    .weight(1)
                )
            )
            .build();
        
        // 注册到游戏
        LootTableManager manager = ...;  // 获取 LootTableManager
        manager.putLootTable(CUSTOM_CHEST, table);
    }
}

7. 数据生成器战利品表

7.1 创建战利品表生成器

package net.example.mymod.datagen;

import net.example.mymod.init.ModBlocks;
import net.example.mymod.init.ModItems;
import net.fabricmc.fabric.api.datagen.v1.FabricDataOutput;
import net.fabricmc.fabric.api.datagen.v1.provider.FabricLootTableProvider;
import net.minecraft.loot.LootTable;
import net.minecraft.loot.LootTables;
import net.minecraft.loot.entry.ItemEntry;
import net.minecraft.loot.entry.LootPoolEntry;
import net.minecraft.loot.function.SetCountLootFunction;
import net.minecraft.loot.provider.number.ConstantLootNumberProvider;
import net.minecraft.loot.provider.number.UniformLootNumberProvider;
import net.minecraft.registry.RegistryKey;

public class ModLootTableProvider extends FabricLootTableProvider {
    
    public ModLootTableProvider(FabricDataOutput output) {
        super(output);
    }
    
    @Override
    public void generate(RegistryKey<LootTable> id) {
        // 根据 ID 分配对应的生成方法
        switch (id.getValue().getPath()) {
            case "blocks/my_block":
                generateMyBlockLoot(id);
                break;
            case "entities/my_entity":
                generateMyEntityLoot(id);
                break;
            case "chests/my_chest":
                generateMyChestLoot(id);
                break;
        }
    }
    
    private void generateMyBlockLoot(RegistryKey<LootTable> id) {
        // 输出生成器
        add(id, LootTable.builder()
            .pool(net.minecraft.loot.LootPool.builder()
                .rolls(ConstantLootNumberProvider.create(1))
                .with(ItemEntry.builder(ModItems.MY_ITEM))
            )
        );
    }
    
    private void generateMyEntityLoot(RegistryKey<LootTable> id) {
        add(id, LootTable.builder()
            .pool(net.minecraft.loot.LootPool.builder()
                .rolls(UniformLootNumberProvider.create(0f, 2f))
                .with(ItemEntry.builder(Items.ROTTEN_FLESH))
            )
            .pool(net.minecraft.loot.LootPool.builder()
                .rolls(ConstantLootNumberProvider.create(1))
                .with(ItemEntry.builder(ModItems.MY_TOKEN)
                    .conditionally(net.minecraft.loot.condition.RandomChanceLootCondition.randomChance(0.1f))
                )
            )
        );
    }
    
    private void generateMyChestLoot(RegistryKey<LootTable> id) {
        add(id, LootTable.builder()
            .pool(net.minecraft.loot.LootPool.builder()
                .rolls(ConstantLootNumberProvider.create(3))
                .with(ItemEntry.builder(Items.GOLD_INGOT)
                    .apply(SetCountLootFunction.builder(UniformLootNumberProvider.create(1f, 5f)))
                )
            )
        );
    }
}

7.2 注册数据生成器

public class MyModDataGenerator implements DataGeneratorEntrypoint {
    @Override
    public void onInitializeDataGenerator(FabricDataGenerator generator) {
        FabricDataGenerator.Pack pack = generator.createPack();
        
        // 添加战利品表生成器
        pack.addProvider(ModLootTableProvider::new);
        
        // 其他生成器
        pack.addProvider(ModRecipeProvider::new);
        pack.addProvider(ModTagProvider::new);
    }
}
