渲染目标、多纹理管理与深度处理

1. 概述

Iris 使用复杂的帧缓冲和纹理系统来管理 GBuffer、中间渲染目标和阴影贴图。

核心文件：

文件路径

`RenderTarget``targets/RenderTarget.java`
`DepthTexture``targets/DepthTexture.java`
`GlFramebuffer``gl/framebuffer/GlFramebuffer.java`
`GlTexture``gl/texture/GlTexture.java`

    // 深度复制策略
    private DepthCopyStrategy copyStrategy;
}

3. 渲染目标管理

3.1 创建渲染目标

public RenderTarget getOrCreate(int index) {
    if (targets[index] != null) return targets[index];
    
    create(index);
    return targets[index];
}

private void create(int index) {
    PackRenderTargetDirectives.RenderTargetSettings settings = targetSettingsMap.get(index);
    
    // 获取缩放尺寸
    Vector2i dimensions = packDirectives.getTextureScaleOverride(index, cachedWidth, cachedHeight);
    
    // 创建渲染目标
    targets[index] = RenderTarget.builder()
        .setDimensions(dimensions.x, dimensions.y)
        .setName("colortex" + index)
        .setInternalFormat(settings.getInternalFormat())
        .setPixelFormat(settings.getPixelFormat())
        .build();
}

3.2 尺寸调整

public boolean resizeIfNeeded(int newDepthBufferVersion, int newDepthTextureId, 
                               int newWidth, int newHeight, 
                               DepthBufferFormat newDepthFormat, 
                               PackDirectives packDirectives) {
    boolean sizeChanged = newWidth != cachedWidth || newHeight != cachedHeight;
    
    // 重新分配深度缓冲
    if (sizeChanged) {
        cachedWidth = newWidth;
        cachedHeight = newHeight;
        
        // 调整所有渲染目标大小
        for (int i = 0; i < targets.length; i++) {
            if (targets[i] != null) {
                targets[i].resize(
                    packDirectives.getTextureScaleOverride(i, newWidth, newHeight));
            }
        }
        
        fullClearRequired = true;
    }
    
    return sizeChanged;
}

4. 深度纹理管理

4.1 DepthTexture

public class DepthTexture {
    private final int textureId;
    private final int width;
    private final int height;
    private final DepthBufferFormat format;
    
    public DepthTexture(String name, int width, int height, DepthBufferFormat format) {
        // 1. 生成纹理
        this.textureId = GlStateManager.genTextures();
        this.width = width;
        this.height = height;
        this.format = format;
        
        // 2. 绑定纹理
        GlStateManager._bindTexture(textureId);
        
        // 3. 分配存储
        GlStateManager._texImage2D(GL_TEXTURE_2D, 0, 
            format.getGlInternalFormat(), width, height, 
            GL_DEPTH_COMPONENT, format.getPixelFormat(), null);
        
        // 4. 设置参数
        GlStateManager.texParameter(GL_TEXTURE_2D, 
            GL_TEXTURE_MIN_FILTER, GL_NEAREST);
        GlStateManager.texParameter(GL_TEXTURE_2D, 
            GL_TEXTURE_MAG_FILTER, GL_NEAREST);
        GlStateManager.texParameter(GL_TEXTURE_2D, 
            GL_TEXTURE_WRAP_S, GL_CLAMP);
        GlStateManager.texParameter(GL_TEXTURE_2D, 
            GL_TEXTURE_WRAP_T, GL_CLAMP);
    }
    
    public int getTextureId() {
        return textureId;
    }
    
    public void resize(int newWidth, int newHeight, DepthBufferFormat newFormat) {
        // 重新分配纹理存储
    }
}

5. GlFramebuffer

5.1 创建帧缓冲

public class GlFramebuffer {
    private final int framebufferId;
    
    public GlFramebuffer() {
        this.framebufferId = GlStateManager.glGenFramebuffers();
    }
    
    public void addColorAttachment(int index, int textureId) {
        GlStateManager._bindFramebuffer(GL_FRAMEBUFFER, framebufferId);
        GlStateManager._FramebufferTexture2D(GL_FRAMEBUFFER, 
            GL_COLOR_ATTACHMENT0 + index, GL_TEXTURE_2D, textureId, 0);
    }
    
    public void addDepthAttachment(int textureId) {
        GlStateManager._bindFramebuffer(GL_FRAMEBUFFER, framebufferId);
        GlStateManager._FramebufferTexture2D(GL_FRAMEBUFFER, 
            GL_DEPTH_ATTACHMENT, GL_TEXTURE_2D, textureId, 0);
    }
    
    public void drawBuffers(int[] buffers) {
        GlStateManager._bindFramebuffer(GL_FRAMEBUFFER, framebufferId);
        GlStateManager._DrawBuffers(buffers);
    }
    
    public void bind() {
        GlStateManager._bindFramebuffer(GL_DRAW_FRAMEBUFFER, framebufferId);
    }
    
    public void bindAsReadBuffer() {
        GlStateManager._bindFramebuffer(GL_READ_FRAMEBUFFER, framebufferId);
    }
    
    public int getStatus() {
        GlStateManager._bindFramebuffer(GL_FRAMEBUFFER, framebufferId);
        return GlStateManager._CheckFramebufferStatus(GL_FRAMEBUFFER);
    }
}

5.2 帧缓冲工厂方法

public GlFramebuffer createGbufferFramebuffer(ImmutableSet<Integer> stageWritesToAlt, 
                                           int[] drawBuffers) {
    if (drawBuffers.length == 0) {
        return createEmptyFramebuffer();
    }
    
    // 计算主缓冲写入集
    ImmutableSet<Integer> stageWritesToMain = invert(stageWritesToAlt, drawBuffers);
    
    // 创建颜色帧缓冲
    GlFramebuffer framebuffer = createColorFramebuffer(stageWritesToMain, drawBuffers);
    
    // 添加深度附件
    framebuffer.addDepthAttachment(currentDepthTexture);
    
    return framebuffer;
}

public GlFramebuffer createColorFramebufferWithDepth(ImmutableSet<Integer> stageWritesToMain, 
                                                   int[] drawBuffers) {
    GlFramebuffer framebuffer = createColorFramebuffer(stageWritesToMain, drawBuffers);
    framebuffer.addDepthAttachment(currentDepthTexture);
    return framebuffer;
}

private GlFramebuffer createColorFramebuffer(ImmutableSet<Integer> stageWritesToMain, 
                                            int[] drawBuffers) {
    GlFramebuffer framebuffer = new GlFramebuffer();
    ownedFramebuffers.add(framebuffer);
    
    for (int i = 0; i < drawBuffers.length; i++) {
        int buffer = drawBuffers[i];
        RenderTarget target = this.getOrCreate(buffer);
        
        // 选择主或备用纹理
        int textureId = stageWritesToMain.contains(buffer) 
            ? target.getMainTexture() 
            : target.getAltTexture();
        
        framebuffer.addColorAttachment(i, textureId);
    }
    
    framebuffer.drawBuffers(actualDrawBuffers);
    
    return framebuffer;
}

6. 缓冲翻转系统

6.1 BufferFlipper

public class BufferFlipper {
    // 记录每个缓冲的翻转状态
    private final Map<Integer, Boolean> flipped = new HashMap<>();
    
    public void flip(int buffer) {
        flipped.compute(buffer, (k, v) -> v == null || !v);
    }
    
    public ImmutableSet<Integer> snapshot() {
        ImmutableSet.Builder<Integer> result = new ImmutableSet.Builder<>();
        
        for (Map.Entry<Integer, Boolean> entry : flipped.entrySet()) {
            if (entry.getValue()) {
                result.add(entry.getKey());
            }
        }
        
        return result.build();
    }
}

7. 纹理缩放

7.1 缩放覆盖

// PackRenderTargetDirectives
public Vector2i getTextureScaleOverride(int index, int width, int height) {
    RenderTargetSettings settings = renderTargetSettings.get(index);
    
    if (settings.hasScaleOverride()) {
        float scale = settings.getScaleOverride();
        return new Vector2i(
            Math.max(1, (int)(width * scale)),
            Math.max(1, (int)(height * scale))
        );
    }
    
    return new Vector2i(width, height);
}

8. 深度复制策略

public interface DepthCopyStrategy {
    void copy(GlFramebuffer source, int sourceDepth, 
              GlFramebuffer dest, int destDepth, 
              int width, int height);
    
    static DepthCopyStrategy fastest(boolean hasStencil) {
        // 根据 OpenGL 版本选择最优策略
        if (IrisRenderSystem.supportsDirectAccess()) {
            return new DirectAccessCopyStrategy(hasStencil);
        } else {
            return new TexImage2DCopyStrategy(hasStencil);
        }
    }
}

public class DirectAccessCopyStrategy implements DepthCopyStrategy {
    @Override
    public void copy(GlFramebuffer source, int sourceDepth, 
                     GlFramebuffer dest, int destDepth, 
                     int width, int height) {
        // 使用 glBlitNamedFramebuffer
        IrisRenderSystem.bindFramebuffer(GL_READ_FRAMEBUFFER, source.getFramebufferId());
        IrisRenderSystem.bindFramebuffer(GL_DRAW_FRAMEBUFFER, dest.getFramebufferId());
        IrisRenderSystem.blitNamedFramebuffer(source.getFramebufferId(), dest.getFramebufferId(),
            0, 0, width, height, 0, 0, width, height,
            GL_DEPTH_BUFFER_BIT, GL_NEAREST);
    }
}
